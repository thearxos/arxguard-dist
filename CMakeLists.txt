cmake_minimum_required(VERSION 3.16)
project(arxguard_native C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
enable_testing()

set(ARXGUARD_NATIVE_SOURCES src/arxguard_engine.c src/arxguard_simd.c)
add_library(arxguard_engine STATIC ${ARXGUARD_NATIVE_SOURCES})
target_include_directories(arxguard_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(arxguard_engine PRIVATE -O3 -fno-plt -fvisibility=hidden)

add_library(arxguard_engine_shared SHARED ${ARXGUARD_NATIVE_SOURCES})
target_include_directories(arxguard_engine_shared PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_options(arxguard_engine_shared PRIVATE -O3 -fno-plt -fvisibility=hidden)
set_target_properties(arxguard_engine_shared PROPERTIES OUTPUT_NAME arxguard_engine)

add_executable(arxguard_native_tests tests/native_engine.c)
target_link_libraries(arxguard_native_tests PRIVATE arxguard_engine)
target_include_directories(arxguard_native_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_test(NAME arxguard_native_vectors COMMAND arxguard_native_tests)

add_executable(arxguard_v4_quality tests/test_engine_v4.c)
target_link_libraries(arxguard_v4_quality PRIVATE arxguard_engine)
target_include_directories(arxguard_v4_quality PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_test(NAME arxguard_v4_quality COMMAND arxguard_v4_quality)

add_executable(arxguard_simd_dispatch_tests tests/test_simd_dispatch.c)
target_link_libraries(arxguard_simd_dispatch_tests PRIVATE arxguard_engine)
target_include_directories(arxguard_simd_dispatch_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_test(NAME arxguard_simd_dispatch COMMAND arxguard_simd_dispatch_tests)

add_executable(arxguard_native_bench bench/native_bench.c)
target_link_libraries(arxguard_native_bench PRIVATE arxguard_engine)
target_compile_options(arxguard_native_bench PRIVATE -O3)

add_executable(arxguard_check tools/arxguard_check.c)
target_link_libraries(arxguard_check PRIVATE arxguard_engine)
target_include_directories(arxguard_check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

find_path(BASH_INCLUDE_DIR builtins.h PATHS /usr/include/bash /usr/local/include/bash)
if(BASH_INCLUDE_DIR)
  add_library(arxguard_native SHARED src/arxguard_builtin.c ${ARXGUARD_NATIVE_SOURCES})
  target_include_directories(arxguard_native PRIVATE ${BASH_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/src)
  target_compile_options(arxguard_native PRIVATE -O3 -fno-plt -fvisibility=hidden)
  set_target_properties(arxguard_native PROPERTIES PREFIX "" OUTPUT_NAME arxguard_native)
endif()

install(TARGETS arxguard_engine_shared LIBRARY DESTINATION lib)
install(TARGETS arxguard_check RUNTIME DESTINATION bin)
install(FILES src/arxguard_engine.h src/arxguard_simd.h DESTINATION include/arxguard)
if(TARGET arxguard_native)
  install(TARGETS arxguard_native LIBRARY DESTINATION lib/arxguard)
endif()
